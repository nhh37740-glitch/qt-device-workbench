cmake_minimum_required(VERSION 3.24)
project(BinaryOnlyWorkbenchClient LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(Qt6 6.8 REQUIRED COMPONENTS Core Widgets)
if(NOT WORKBENCH_SDK)
  message(FATAL_ERROR "Set WORKBENCH_SDK to delivery package shared/sdk")
endif()
add_executable(binary-consumer binary_consumer.cpp)
target_include_directories(binary-consumer PRIVATE "${WORKBENCH_SDK}/include")
target_link_libraries(binary-consumer PRIVATE "${WORKBENCH_SDK}/lib/wb_contracts.lib" Qt6::Core Qt6::Widgets)
if(MSVC)
  target_compile_options(binary-consumer PRIVATE /utf-8)
endif()
