cmake_minimum_required(VERSION 3.24)
project(QtDeviceWorkbench VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTOUIC ON)
set(CMAKE_AUTORCC ON)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
set(CMAKE_ARCHIVE_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/lib")
foreach(config Debug Release RelWithDebInfo MinSizeRel)
  string(TOUPPER "${config}" uc)
  set(CMAKE_RUNTIME_OUTPUT_DIRECTORY_${uc} "${CMAKE_BINARY_DIR}/bin")
  set(CMAKE_LIBRARY_OUTPUT_DIRECTORY_${uc} "${CMAKE_BINARY_DIR}/bin")
  set(CMAKE_ARCHIVE_OUTPUT_DIRECTORY_${uc} "${CMAKE_BINARY_DIR}/lib")
endforeach()
if(MSVC)
  add_compile_options(/utf-8 /W4 /permissive-)
endif()
find_package(Qt6 6.8 REQUIRED COMPONENTS Core Network Widgets Test)
include(CTest)
include_directories("${PROJECT_SOURCE_DIR}/include")
add_library(wb_contracts SHARED contracts/contracts.cpp include/workbench/contracts.h)
target_compile_definitions(wb_contracts PRIVATE WORKBENCH_CONTRACTS_BUILD)
target_include_directories(wb_contracts PUBLIC "${PROJECT_SOURCE_DIR}/include")
target_link_libraries(wb_contracts PUBLIC Qt6::Core Qt6::Widgets)
foreach(module wire_protocol device_simulator device_source frontend record_store result_push result_receiver)
  add_subdirectory(modules/${module})
endforeach()
add_subdirectory(apps/simulator)
add_subdirectory(apps/receiver)
add_executable(device-workbench apps/workbench/main.cpp)
target_link_libraries(device-workbench PRIVATE wb_contracts wb_device_source wb_frontend wb_record_store wb_result_push Qt6::Network)
if(BUILD_TESTING)
  foreach(module contracts wire_protocol device_simulator device_source frontend record_store result_push result_receiver)
    add_executable(test_${module} tests/${module}_test.cpp)
    target_link_libraries(test_${module} PRIVATE wb_contracts Qt6::Test Qt6::Network Qt6::Widgets)
    if(NOT module STREQUAL "contracts")
      target_link_libraries(test_${module} PRIVATE wb_${module})
    endif()
    if(module STREQUAL "device_simulator" OR module STREQUAL "device_source" OR module STREQUAL "result_push" OR module STREQUAL "result_receiver")
      target_link_libraries(test_${module} PRIVATE wb_wire_protocol)
    endif()
    if(module STREQUAL "result_push")
      target_link_libraries(test_${module} PRIVATE wb_result_receiver)
    endif()
    add_test(NAME ${module} COMMAND test_${module})
    set_tests_properties(${module} PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 90)
  endforeach()
endif()
install(TARGETS wb_contracts wb_wire_protocol wb_device_simulator wb_device_source wb_frontend wb_record_store wb_result_push wb_result_receiver device-simulator device-workbench result-receiver RUNTIME DESTINATION bin LIBRARY DESTINATION bin ARCHIVE DESTINATION sdk/lib)
install(FILES include/workbench/contracts.h DESTINATION sdk/include/workbench)
install(DIRECTORY docs/ DESTINATION docs)
install(DIRECTORY examples/ DESTINATION examples)
install(FILES README.md DESTINATION .)
